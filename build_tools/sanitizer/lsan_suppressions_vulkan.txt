# LSAN suppressions for Vulkan driver leaks.
#
# Mesa/RADV worker threads initialize a global CPU topology table through
# pthread_once. That allocation can remain allocated after driver unload.

leak:libvulkan_radeon.so
leak:libvulkan_lvp.so

# Vulkan loader physical-device terminator setup may allocate ICD bookkeeping
# that remains reachable only through loader-internal process lifetime state.
leak:setup_loader_term_phys_devs

# Unloaded driver frames may have no module name available for matching. This
# fallback also matches non-driver pthread_once allocations and is broader than
# the module rules above.
leak:__pthread_once_slow
